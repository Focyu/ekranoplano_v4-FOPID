function u= fopid_altura_mod(e, y, kp, ki, kd)

%#codegen
% FOPI^lambda D discreto - ALTURA
% e : error de altura
% y : medicion de altura
% u : referencia de pitch (theta_sp)

mu     = 0.9;
lambda = 0.7;
Ts     = 0.01;
wc     = 1.0;
u_min  = -0.2;
u_max  =  0.2;

alpha = wc / tan(wc * Ts / 2);

KP = kp;
KD = kd * (alpha^mu);
KI = ki * (alpha^(-lambda));

f0 = 1;
f1 = -2*mu;
f2 = 2*mu^2;
f3 = (-4*mu^3)/3 - (2*mu)/3;
f4 = (2*mu^4)/3 + (4*mu^2)/3;
f5 = (-4*mu^5)/15 - (4*mu^3)/3 - (2*mu)/5;

lm = 1 - lambda;
g0 = 1;
g1 = -2*lm;
g2 = 2*lm^2;
g3 = (-4*lm^3)/3 - (2*lm)/3;
g4 = (2*lm^4)/3 + (4*lm^2)/3;
g5 = (-4*lm^5)/15 - (4*lm^3)/3 - (2*lm)/5;

% Actualizacion de historias


persistent e1 e2 e3 e4 e5 e6 e7
persistent y1 y2 y3 y4 y5 y6 y7
persistent sumI u_prev u_prev2
% 
if isempty(e1)
    e1 = 0; e2 = 0; e3 = 0; e4 = 0; e5 = 0; e6 = 0; e7 = 0;
    y1 = 0; y2 = 0; y3 = 0; y4 = 0; y5 = 0; y6 = 0; y7 = 0;
    sumI = 0;
    u_prev = 0;
    u_prev2 = 0;
end

% 
% 
% e7 = e6; e6 = e5; e5 = e4; e4 = e3; e3 = e2; e2 = e1; e1 = e;
% y7 = y6; y6 = y5; y5 = y4; y4 = y3; y3 = y2; y2 = y1; y1 = y;

% Derivada fraccional sobre la medicion
sumD = KD * ( ...
    f0*(y  - 2*y1 + y2) ...
    + f1*(y1 - 2*y2 + y3) ...
    + f2*(y2 - 2*y3 + y4) ...
    + f3*(y3 - 2*y4 + y5) ...
    + f4*(y4 - 2*y5 + y6) ...
    + f5*(y5 - 2*y6 + y7) );



% Integral fraccional sobre el error
sumI = KI * ( ...
    g0*(e  + 2*e1 + e2) ...
    + g1*(e1 + 2*e2 + e3) ...
    + g2*(e2 + 2*e3 + e4) ...
    + g3*(e3 + 2*e4 + e5) ...
    + g4*(e4 + 2*e5 + e6) ...
    + g5*(e5 + 2*e6 + e7) );



% Salida final
% u_raw = u_prev2 + KP*(e - e2) + sumD + sumI;
% u = max(min(u_raw, u_max), u_min);
% 
% 
% u_prev2 = u_prev;
% u_prev  = u;
u_raw = u_prev + KP*(e - e1) + sumD + sumI;
u = max(min(u_raw, u_max), u_min);

u_prev2 = u_prev;
u_prev  = u;

e7 = e6; e6 = e5; e5 = e4; e4 = e3; e3 = e2; e2 = e1; e1 = e;
y7 = y6; y6 = y5; y5 = y4; y4 = y3; y3 = y2; y2 = y1; y1 = y;

end

function u= fopid_roll_mod(e, y, kp, ki, kd)

%#codegen
% FOPI^lambda D discreto - ALTURA
% e : error de altura
% y : medicion de altura
% u : referencia de pitch (theta_sp)

mu     = 0.9;
lambda = 0.7;
Ts     = 0.01;
wc     = 1.0;
u_min  = -0.1;
u_max  =  0.1;

alpha = wc / tan(wc * Ts / 2);

KP = kp;
KD = kd * (alpha^mu);
KI = ki * (alpha^(-lambda));

f0 = 1;
f1 = -2*mu;
f2 = 2*mu^2;
f3 = (-4*mu^3)/3 - (2*mu)/3;
f4 = (2*mu^4)/3 + (4*mu^2)/3;
f5 = (-4*mu^5)/15 - (4*mu^3)/3 - (2*mu)/5;

lm = 1 - lambda;
g0 = 1;
g1 = -2*lm;
g2 = 2*lm^2;
g3 = (-4*lm^3)/3 - (2*lm)/3;
g4 = (2*lm^4)/3 + (4*lm^2)/3;
g5 = (-4*lm^5)/15 - (4*lm^3)/3 - (2*lm)/5;

% Actualizacion de historias


persistent e1 e2 e3 e4 e5 e6 e7
persistent y1 y2 y3 y4 y5 y6 y7
persistent sumI u_prev u_prev2
% 
if isempty(e1)
    e1 = 0; e2 = 0; e3 = 0; e4 = 0; e5 = 0; e6 = 0; e7 = 0;
    y1 = 0; y2 = 0; y3 = 0; y4 = 0; y5 = 0; y6 = 0; y7 = 0;
    sumI = 0;
    u_prev = 0;
    u_prev2 = 0;
end

% 
% 
% e7 = e6; e6 = e5; e5 = e4; e4 = e3; e3 = e2; e2 = e1; e1 = e;
% y7 = y6; y6 = y5; y5 = y4; y4 = y3; y3 = y2; y2 = y1; y1 = y;

% Derivada fraccional sobre la medicion
sumD = KD * ( ...
    f0*(y  - 2*y1 + y2) ...
    + f1*(y1 - 2*y2 + y3) ...
    + f2*(y2 - 2*y3 + y4) ...
    + f3*(y3 - 2*y4 + y5) ...
    + f4*(y4 - 2*y5 + y6) ...
    + f5*(y5 - 2*y6 + y7) );



% Integral fraccional sobre el error
sumI = KI * ( ...
    g0*(e  + 2*e1 + e2) ...
    + g1*(e1 + 2*e2 + e3) ...
    + g2*(e2 + 2*e3 + e4) ...
    + g3*(e3 + 2*e4 + e5) ...
    + g4*(e4 + 2*e5 + e6) ...
    + g5*(e5 + 2*e6 + e7) );

% Salida final

u_raw = u_prev + KP*(e - e1) + sumD + sumI;
u = max(min(u_raw, u_max), u_min);

u_prev2 = u_prev;
u_prev  = u;

e7 = e6; e6 = e5; e5 = e4; e4 = e3; e3 = e2; e2 = e1; e1 = e;
y7 = y6; y6 = y5; y5 = y4; y4 = y3; y3 = y2; y2 = y1; y1 = y;

end

function [XDOT, Va_out, CL_total, h_out, mu_Lw_out, mu_Dw_out] = Model(X, U, U_wind, W_wind)
% Modelo Dinámico del Ekranoplano AirFish 8 (Sincronizado)
% ESCALA: 1:21.5 | Soporte Coder y Viento
% CORRECCIONES v2: D_prop 3.5"→5", alpha_0h, iw, CD_0
% CORRECCIONES v3: Desacoplamiento Ixz (feedforward interno roll-yaw)
% CORRECCIONES v4: N_dec simétrico, C_D_e unidades, signo motor Fz
%#codegen
coder.varsize('XDOT',      [40, 1], [0, 0]);
coder.varsize('Va_out',    [1,  1], [0, 0]);
coder.varsize('CL_total',  [1,  1], [0, 0]);
coder.varsize('h_out',     [1,  1], [0, 0]);
coder.varsize('mu_Lw_out', [1,  1], [0, 0]);
coder.varsize('mu_Dw_out', [1,  1], [0, 0]);

%-----------STATE AND CONTROL VECTORS-----------------------------
x1=X(1); x2=X(2); x3=X(3); x4=X(4); x5=X(5); x6=X(6);
x7=X(7); x8=X(8); x9=X(9); x10=X(10); x11=X(11); x12=X(12);
u1=U(1); u2=U(2); u3=U(3); u4=U(4); u5=U(5);

%-----------SYSTEM PARAMETERS (ESCALA 1:21.5)---------------------
m = 1.200; bw = 0.6977; bh = 0.3808; cbar = 0.0930; lt = 0.5600;
Sw = 0.0649; Sh = 0.02164; ARw = 7.5; ARh = 6.7; TRw = 0.4; TRh = 0.5;
Xcg = 0; Ycg = 0; Zcg = 0; Xac = 0.0135; Yac = 0; Zac = -0.0083;
Xapt1 = -0.0721; Yapt1 =  0.0834; Zapt1 = 0.0396;
Xapt2 = -0.0721; Yapt2 = -0.0834; Zapt2 = 0.0396;
rho = 1.225; g = 9.81; e0w = 0.9; e0h = 0.9;
alpha_0w = -3.75*pi/180;
alpha_0h = -2.50*pi/180;
iw = 1.5*pi/180;
ih = 0.5*pi/180;
zw = 0.0505; zh = 0.3475;
CD_0 = 0.0420;

%-----------SATURACIÓN DE CONTROLES-------------------------------
u1 = max(min(u1,  20*pi/180), -20*pi/180);
u2 = max(min(u2,  20*pi/180), -20*pi/180);
u3 = max(min(u3,  15*pi/180), -15*pi/180);
u4 = max(min(u4, 1.0), 0.0);
u5 = max(min(u5, 1.0), 0.0);

%-----------VIENTO Y VELOCIDAD RELATIVA---------------------------
u_wg = U_wind(1); v_wg = U_wind(2); w_wg = U_wind(3);
p_wg = W_wind(1); q_wg = W_wind(2); r_wg = W_wind(3);

u_r = x1 + u_wg;
v_r = x2 + v_wg;
w_r = x3 + w_wg;

Va = sqrt(u_r^2 + v_r^2 + w_r^2);
if Va < 0.5
    Va    = 0.5;
    alpha = 0.0;
    beta  = 0.0;
else
    alpha = atan2(w_r, u_r);
    beta  = asin(max(-1.0, min(1.0, v_r / Va)));
end
Va_out = Va;

p_aero = x4 + p_wg;
q_aero = x5 + q_wg;
r_aero = x6 + r_wg;

h  = -x12;
hw = max(0.001, h - zw);
hh = max(0.001, h + zh);
Q     = 0.5*rho*Va^2;
wbe_b = [x4; x5; x6];
V_b   = [x1; x2; x3];

%-----------EFECTO SUELO------------------------------------------
mu_Lw = 1 + (1 - 2.25*(TRw^0.00273 - 0.997)*((ARw^0.717)+13.6)) * ...
        (288*abs(hw/bw)^0.787*exp(-9.14*(abs(hw/bw)^0.327))) / (ARw^0.882);
mu_Lh = 1 + (1 - 2.25*(TRh^0.00273 - 0.997)*((ARh^0.717)+13.6)) * ...
        (288*abs(hh/bh)^0.787*exp(-9.14*(abs(hh/bh)^0.327))) / (ARh^0.882);

mu_Dw = 1 - (1 - 0.157*max(0,(TRw^0.757-0.373))*max(0,(ARw^0.417-1.27))) * ...
        exp(-4.74*max(0,abs(hw/bw)^0.814)) - abs(hw/bw)^2*exp(-3.88*max(0,abs(hw/bw)^0.758));
mu_Dh = 1 - (1 - 0.157*max(0,(TRh^0.757-0.373))*max(0,(ARh^0.417-1.27))) * ...
        exp(-4.74*max(0,abs(hh/bh)^0.814)) - abs(hh/bh)^2*exp(-3.88*max(0,abs(hh/bh)^0.758));

%-----------AERODINÁMICA------------------------------------------
deps_dalpha = 0.35;
eps = deps_dalpha*(alpha - alpha_0w) + deps_dalpha*(lt/Va)*q_aero;

CL_w_OGE = 2*pi*(ARw/(ARw+2))*(alpha - alpha_0w + iw);
CL_h_OGE = 2*pi*(ARh/(ARh+2))*(alpha - alpha_0h + ih - eps);
CL_w_IGE  = CL_w_OGE * mu_Lw;
CL_h_IGE  = CL_h_OGE * mu_Lh;

CD_iw_IGE = (CL_w_IGE^2)/(pi*ARw*e0w) * mu_Dw;
CD_ih_IGE = (CL_h_IGE^2)/(pi*ARh*e0h) * mu_Dh;

% v4: u2 a grados para la polar de arrastre del elevador
u2_deg = u2 * 180/pi;
C_D_e  = -0.0000108*u2_deg^2 + 0.000715*u2_deg;           % <<< CORREGIDO v4

Dtot = Q*(CD_0*Sw + CD_iw_IGE*Sw + CD_ih_IGE*Sh + C_D_e*Sh);
Ltot = Q*(CL_w_IGE*Sw + CL_h_IGE*Sh);
CQ   = -0.019*beta*180/pi;

FA_s = [-Dtot; CQ*Q*Sw; -Ltot];
C_bs = [cos(alpha) 0 -sin(alpha); 0 1 0; sin(alpha) 0 cos(alpha)];
FA_b = C_bs * FA_s;

%-----------MOMENTOS AERODINÁMICOS-------------------------------
Cm_alpha = -1.14; Cm_q = -5.0; Cm_de = -3.0; Cm_h = 0.05;
Delta_Cm_IGE = -Cm_h * exp(-4.0*abs(hw/bw));
Cl_da = -0.5; Cn_dr = -0.3; Cl_p = -2.0; Cn_r = -1.5;
Cn_da = -0.05;

Cl_beta = -0.12;
Cl = Cl_beta*beta + Cl_p*(p_aero*bw/(2*Va)) + Cl_da*u1;
Cm = Cm_alpha*alpha + Delta_Cm_IGE + Cm_q*(q_aero*cbar/(2*Va)) + Cm_de*u2;
Cn = -0.1146*beta   + Cn_r*(r_aero*bw/(2*Va))   + Cn_dr*u3  + Cn_da*u1;
MAcg_b = [Cl; Cm; Cn] .* [bw*Q*Sw; cbar*Q*Sw; bw*Q*Sw];

%-----------MOTOR BRUSHLESS---------------------------
km_motor = 32.0;
Cp_motor = 0.57;
D_prop   = 0.1270;
Sp_prop  = pi*(D_prop^2/4);

Vd1 = Va + u4*(km_motor - Va);
Vd2 = Va + u5*(km_motor - Va);
Tp1 = 0.5*rho*Sp_prop*Cp_motor*Vd1*(Vd1 - Va);
Tp2 = 0.5*rho*Sp_prop*Cp_motor*Vd2*(Vd2 - Va);

% v4: componente Fz negativa (hacia arriba en body frame) para inclinación nose-up
FE1_b = [Tp1*cos(5*pi/180); 0; -Tp1*sin(5*pi/180)];       % <<< CORREGIDO v4
FE2_b = [Tp2*cos(5*pi/180); 0; -Tp2*sin(5*pi/180)];       % <<< CORREGIDO v4
FE_b  = FE1_b + FE2_b;

MEcg1_b = cross([Xcg-Xapt1; Yapt1-Ycg; Zcg-Zapt1], FE1_b);
MEcg2_b = cross([Xcg-Xapt2; Yapt2-Ycg; Zcg-Zapt2], FE2_b);
MEcg_b  = MEcg1_b + MEcg2_b;

%-----------DINÁMICA---------------------------------------------
g_b  = [-g*sin(x8); g*cos(x8)*sin(x7); g*cos(x8)*cos(x7)];
Fg_b = m*g_b;
F_b  = Fg_b + FE_b + FA_b;
x1to3dot = (1/m)*F_b - cross(wbe_b, V_b);
Mcg_b = MAcg_b + MEcg_b;

Ixx = 0.008217; Iyy = 0.017695; Izz = 0.023669; Ixz = 0.001856;
Gamma = Ixx*Izz - Ixz^2;
p = wbe_b(1); q = wbe_b(2); r = wbe_b(3);
L = Mcg_b(1); M = Mcg_b(2); N = Mcg_b(3);

% --- DESACOPLAMIENTO Ixz SIMÉTRICO (v4) --------------------------
L_dec = L - (Ixz / Izz) * N;                               % <<< CORREGIDO v4
N_dec = N - (Ixz / Ixx) * L;                               % <<< NUEVO v4
% -----------------------------------------------------------------

p_dot = (Izz*L_dec + Ixz*N_dec - (Ixz*(Iyy-Ixx-Izz)*p + (Ixz^2+Izz*(Izz-Iyy))*r)*q) / Gamma;
q_dot = (M - (Ixx-Izz)*p*r - Ixz*(p^2-r^2)) / Iyy;
r_dot = (Ixz*L_dec + Ixx*N_dec + (Ixz*(Iyy-Ixx-Izz)*r + (Ixz^2+Ixx*(Ixx-Iyy))*p)*q) / Gamma;

x4to6dot = [p_dot; q_dot; r_dot];

H_phi = [1 sin(x7)*tan(x8) cos(x7)*tan(x8);
         0 cos(x7)         -sin(x7);
         0 sin(x7)/cos(x8)  cos(x7)/cos(x8)];
x7to9dot = H_phi * wbe_b;

x10to12dot = rot_body_to_ned(X(1:9));

%-----------SALIDAS----------------------------------------------
LD_ratio = Ltot / max(Dtot, 0.001);
XDOT = [x1to3dot; x4to6dot; x7to9dot; x10to12dot; ...
        LD_ratio; F_b; Mcg_b; CQ; Cl; Cm; Cn; alpha; beta; ...
        CL_w_OGE; CL_h_OGE; CL_w_IGE; CL_h_IGE; ...
        CD_iw_IGE; CD_ih_IGE; Fg_b; FE_b; FA_b];

CL_total   = (CL_w_IGE*Sw + CL_h_IGE*Sh) / (Sw + Sh);
h_out      = -x12;
mu_Lw_out  = mu_Lw;
mu_Dw_out  = mu_Dw;
end

function [V_NED] = rot_body_to_ned(X)

c_the = cos(X(8)); s_the = sin(X(8));
c_psi = cos(X(9)); s_psi = sin(X(9));

end

% BLOQUE 1: FLOTABILIDAD CONTINUA (sustituye a "MATLAB Function-reset")
% ============================================================
function [Fz_boya, x_reset] = fcn_flotabilidad(x, h_max, k_boya, c_boya)
%#codegen
% Fuerza de flotabilidad + amortiguamiento, activa solo cerca del agua.
%
% Entradas:
%   x       : vector de 12 estados [u v w p q r phi theta psi x y z_NED]
%   h_max   : altura límite superior de la "zona de flotación" (0.5 m)
%   k_boya  : rigidez de la fuerza restauradora (N/m) - ajustar en pruebas
%   c_boya  : amortiguamiento (N·s/m) - evita oscilaciones tipo resorte
%
% Salidas:
%   Fz_boya : fuerza vertical [N] a SUMAR en Fbz/Fgz dentro del Model()
%   x_reset : estado "limpio" opcional (solo para telemetría/anti-windup),
%             NO se usa como reset duro del Integrator.
%
% Física:
%   h_actual = -z_NED (convención NED: z negativo = altura positiva)
%   Si 0 <= h_actual <= h_max:
%       Fz_boya = -k_boya*(h_actual - h_ref) - c_boya*(-w)
%       (like Arquimedes: entre más se "hunde" bajo h_ref, más empuja
%        hacia arriba; el amortiguador frena el rebote para que no
%        oscile indefinidamente)
%   Si h_actual < 0 (penetró el agua): fuerza de empuje fuerte (piso duro suave)
%   Si h_actual > h_max: sin fuerza de flotación (vuelo libre en efecto suelo)

h_ref    = h_max/2;      % nivel de equilibrio deseado (~0.25 m)
h_actual = -x(12);
w        = x(3);         % velocidad vertical en ejes cuerpo (+ = bajando)

x_reset = x;             % no se usa para reset; se deja igual

if h_actual <= h_max
    % Zona de flotación: resorte + amortiguador, sin discontinuidad
    Fz_boya = -k_boya*(h_actual - h_ref) - c_boya*(-w);

    % Limitador de seguridad: evita que la fuerza se dispare si
    % h_actual se vuelve muy negativo por overshoot numérico
    Fz_boya = max(min(Fz_boya, 3*k_boya*h_max), -3*k_boya*h_max);
else
    Fz_boya = 0.0;  % fuera de la zona de flotación, sin fuerza extra
end

end
